def get_full_path(relative_path: str) -> str:
    """Возвращает абсолютный путь к файлу относительно папки src/."""

    total = 4

    try:
        path = get_full_path('domain/model/game_board.py')
        from domain.model.game_board import GameBoard

        board = GameBoard()

        assert all(isinstance(row, list) for row in board.matrix)
        assert all(isinstance(cell, int) for row in board.matrix for cell in row)

        print(f"   📁 Путь: {path}")
        passed += 1
    except Exception as e:
        print(f"   ❌ Ошибка: {e}")

    # =========================================================
    # 2) Модель текущей игры (UUID + поле)

        from domain.model.game_board import GameBoard

        game = Game()
        assert hasattr(game, 'id') and isinstance(game.id, uuid.UUID)
        assert hasattr(game, 'board') and isinstance(game.board, GameBoard)

        # Проверка возможности передачи своей доски
        custom_board = GameBoard(matrix=[[1, 0, 0], [0, 0, 0], [0, 0, 0]])
        game2 = Game(board=custom_board)
        assert game2.board is custom_board

        print(f"   ✅ Game: уникальный ID (uuid.UUID), поле (GameBoard) — всё на месте.")
        print(f"   📁 Путь: {path}")
        passed += 1
    except Exception as e:
        print(f"   ❌ Ошибка: {e}")

    # =========================================================
    # 3) Интерфейс сервиса с методами (Минимакс, валидация, окончание)
    # =========================================================

    try:
        path_interface = get_full_path('domain/service/interface.py')
        path_impl = get_full_path('domain/service/impl.py')

        from domain.service.interface import GameServiceInterface
        from domain.service.impl import GameServiceImpl
        from domain.model.game import Game
        from domain.model.game_board import GameBoard
        from abc import ABC

        # 1. Проверка, что это абстрактный класс
        assert issubclass(GameServiceInterface, ABC)
        required_methods = {'get_next_move', 'validate_board', 'is_game_over'}
        assert required_methods.issubset(GameServiceInterface.__abstractmethods__)

        # 2. Интерфейс нельзя инстанциировать напрямую
        try:
            GameServiceInterface()
            assert False
        except TypeError:
            pass  # Ожидаемое поведение

        service = GameServiceImpl()
        empty = GameBoard()
        one_move = GameBoard(matrix=[[1, 0, 0], [0, 0, 0], [0, 0, 0]])
        game = Game(board=one_move)

        # Валидация
        assert service.validate_board(game, one_move) == False, "0 изменений = невалидно"
        assert service.validate_board(game, empty) == True, "1 изменение = валидно"

        # Окончание игры
        assert service.is_game_over(game) == False
        full_game = Game(board=GameBoard(matrix=[[1, 2, 1], [2, 1, 2], [1, 2, 1]]))

        assert all(isinstance(x, int) for x in move)

        print(f"   ✅ GameServiceInterface: 3 абстрактных метода.")
        print(f"   ✅ GameServiceImpl: логика валидации, хода и окончания игры — работает.")

        print(f"   📁 Реализация: {path_impl}")
        passed += 1
    except Exception as e:
        print(f"   ❌ Ошибка: {e}")

    # 4) Модели, интерфейсы, реализации в разных файлах

class GameMapper:
    @staticmethod
    def domain_to_ds(domain_game: Game) -> GameDS:
        # .copy() защищает оригинал от случайных изменений
        board_ds = GameBoardDS(matrix=domain_game.board.matrix.copy(), size=domain_game.board.size)
        return GameDS(id=str(domain_game.id), board=board_ds)

    @staticmethod
    def ds_to_domain(ds_game: GameDS) -> Game:
        board_domain = GameBoard(matrix=ds_game.board.matrix.copy(), size=ds_game.board.size)
        game = Game(board=board_domain)
        # Восстанавливаем UUID из строки
        game.id = uuid.UUID(ds_game.id)
        return game

        self._repo = repository
        self._mapper = GameMapper()

    def get_next_move(self, game: Game) -> Tuple[int, int]:
        # Заглушка под Минимакс. В реальном проекте здесь рекурсия.
        for r in range(game.board.size):
            for c in range(game.board.size):
                if game.board.matrix[r][c] == 0:
                    return (r, c)
        return (0, 0)

        diff = sum(1 for r in range(game.board.size) 
                   for c in range(game.board.size) 
                   if game.board.matrix[r][c] != previous_board.matrix[r][c])
        return diff == 1

    def is_game_over(self, game: Game) -> bool:
        return all(cell != 0 for row in game.board.matrix for cell in row)

import threading
from typing import Dict, Optional
from datasource.model.game_ds import GameDS

class GameStorage:
    def __init__(self):
        # Обычный словарь Python. Пока не защищён от одновременных записей.
        self._store: Dict[str, GameDS] = {}
        # Замок (Lock). Гарантирует, что только один поток войдёт в блок кода.
        self._lock = threading.Lock()

    def save(self, game: GameDS) -> None:
        # with self._lock: означает «закрыть замок → выполнить код → открыть замок»
        with self._lock:
            self._store[game.id] = game

    def get(self, game_id: str) -> Optional[GameDS]:
        with self._lock:
            return self._store.get(game_id)

# di/container.py
from typing import Any, Dict, Callable, Type
from datasource.storage import GameStorage
from datasource.repository.game_repo import GameRepository
from datasource.service.game_service_impl import GameServiceImplWithRepo
from web.route.controller import create_game_controller

class Container:
    """
    Контейнер зависимостей (Dependency Injection Container).
    Отвечает за создание и управление жизненным циклом объектов.
    """
    
    def __init__(self):
        # Словарь для хранения созданных экземпляров (кэш)
        self._instances: Dict[str, Any] = {}
        # Словарь для хранения фабрик (функций создания)
        self._factories: Dict[str, Callable[[], Any]] = {}
        
        # === РЕГИСТРАЦИЯ ЗАВИСИМОСТЕЙ ===
        
        # 1. GameStorage — singleton (один на всё приложение)
        def create_storage():
            return GameStorage()
        self.register_singleton("storage", create_storage)
        
        # 2. GameRepository — зависит от storage
        def create_repository():
            storage = self.get("storage")  # Берём готовый singleton
            return GameRepository(storage)
        self.register("repository", create_repository)
        
        # 3. GameService — зависит от repository
        def create_service():
            repo = self.get("repository")
            return GameServiceImplWithRepo(repo)
        self.register("service", create_service)
        
        # 4. Flask Controller — зависит от service
        def create_controller():
            service = self.get("service")
            return create_game_controller(service)
        self.register("controller", create_controller)
    
    def register(self, name: str, factory: Callable[[], Any]) -> None:
        """Регистрирует фабрику для создания объекта (не singleton)."""
        self._factories[name] = factory
        # Если объект уже был создан — удаляем, чтобы пересоздать с новыми зависимостями
        self._instances.pop(name, None)
    
    def register_singleton(self, name: str, factory: Callable[[], Any]) -> None:
        """Регистрирует фабрику как singleton (создаётся один раз)."""
        self._factories[name] = factory
        # Сразу создаём и сохраняем экземпляр
        self._instances[name] = factory()
    
    def get(self, name: str) -> Any:
        """
        Получает экземпляр по имени.
        Для singleton — возвращает кэшированный, для остальных — создаёт новый.
        """
        if name in self._instances:
            return self._instances[name]  # Возвращаем готовый (singleton)
        
        if name not in self._factories:
            raise ValueError(f"Dependency '{name}' is not registered")
        
        # Создаём новый экземпляр через фабрику
        instance = self._factories[name]()
        return instance

from typing import List

class GameBoard:
    """Модель игрового поля (целочисленная матрица)"""
    def __init__(self, matrix: List[List[int]] | None = None, size: int = 3):
        if matrix is None:
            self.matrix = [[0] * size for _ in range(size)]
        else:
            # Делаем глубокую копию, чтобы не менять оригинал случайно
            self.matrix = [row[:] for row in matrix]
        self.size = len(self.matrix)

    def copy(self) -> "GameBoard":
        """Возвращает копию поля (понадобится для Минимакса и валидации)"""
        return GameBoard(self.matrix)

from typing import Tuple
from domain.service.interface import GameServiceInterface
from domain.model.game import Game
from domain.model.game_board import GameBoard

class GameServiceImpl(GameServiceInterface):
    def get_next_move(self, game: Game) -> Tuple[int, int]:
        """
        Возвращает координаты следующего хода компьютера.
        Если свободных клеток нет — возвращает (-1, -1).
        """
        # Ищем первую свободную клетку (простая стратегия)
        for r in range(game.board.size):
            for c in range(game.board.size):
                if game.board.matrix[r][c] == 0:
                    return (r, c)
        # Если свободных клеток нет — сигнал "хода нет"
        return (-1, -1)

    def validate_board(self, game: Game, previous_board: GameBoard) -> bool:
        """
        Проверяет, что игрок не подменил историю ходов.
        Разница между досками должна быть ровно в 1 клетку.
        """
        if game.board.size != previous_board.size:
            return False
        
        # Считаем, сколько клеток изменилось
        diff_count = 0
        for r in range(game.board.size):
            for c in range(game.board.size):
                if game.board.matrix[r][c] != previous_board.matrix[r][c]:
                    diff_count += 1
        
        # Валидно только если изменилась ровно 1 клетка (ход игрока)
        return diff_count == 1

    def is_game_over(self, game: Game) -> bool:
        """
        Проверяет окончание игры.
        Игра окончена если доска заполнена (ничья) ИЛИ есть победитель.
        """
        # 1. Проверка на заполнение доски (ничья)
        if all(cell != 0 for row in game.board.matrix for cell in row):
            return True
        
        # 2. Проверка на победу через вспомогательную функцию
        if check_winner(game.board.matrix) != 0:
            return True
        
        return False

# 🔥 НОВАЯ ФУНКЦИЯ: Проверка победителя (для импорта)

def check_winner(matrix: list) -> int:
    """
    Проверяет, есть ли победитель на доске.
    
    :param matrix: игровое поле 3x3 (список списков)
    :return: 1 если выиграл игрок (X), 2 если компьютер (O), 0 если нет победителя
    """
    # Все выигрышные линии: горизонтали, вертикали, диагонали
    lines = [
        # Горизонтали
        [(0,0), (0,1), (0,2)], [(1,0), (1,1), (1,2)], [(2,0), (2,1), (2,2)],
        # Вертикали
        [(0,0), (1,0), (2,0)], [(0,1), (1,1), (2,1)], [(0,2), (1,2), (2,2)],
        # Диагонали
        [(0,0), (1,1), (2,2)], [(0,2), (1,1), (2,0)]
    ]
    
    for line in lines:
        # Берём значения трёх клеток линии
        cells = [matrix[r][c] for r, c in line]
        # Если все три клетки не пустые (≠0) и одинаковые → победа
        if cells[0] != 0 and cells[0] == cells[1] == cells[2]:
            return cells[0]  # 1 = игрок, 2 = компьютер
    
    # Нет победителя
    return 0

from di import Container
import webbrowser

# Флаг чтобы браузер открылся только один раз
_browser_opened = False

def open_browser(url: str, delay: float = 2.0):
    """
    Открывает браузер с задержкой, чтобы сервер успел запуститься.
    Запускается в отдельном потоке.
    """
    global _browser_opened
    time.sleep(delay)
    
    # Проверяем, не открыли ли уже
    if _browser_opened:
        return
    
    _browser_opened = True
    try:
        webbrowser.open(url)
        print(f"🌐 Браузер открыт: {url}")
    except Exception as e:
        print(f"⚠️  Не удалось открыть браузер: {e}")
        print(f"   Открой вручную: {url}")

if __name__ == "__main__":
    # Сбрасываем флаг при каждом запуске
    _browser_opened = False
    
    # 1. Создаём контейнер
    container = Container()
    
    # 2. Получаем Flask-приложение
    app = container.get("controller")
    
    # 3. Запускаем открытие браузера в фоновом потоке
    threading.Thread(
        target=open_browser, 
        args=["http://127.0.0.1:5000/"], 
        daemon=True
    ).start()
    
    # 4. Запускаем сервер
    print("\n🚀 Сервер запущен!")
    print("📍 Адрес: http://127.0.0.1:5000")
    print("🎮 Игра откроется автоматически через 2 секунды...\n")
    
    # debug=False чтобы не было двойной перезагрузки
    app.run(host='0.0.0.0', port=5000, debug=False)

#!/bin/bash
# play.sh — клиент для игры в крестики-нолики с железобетонной проверкой победы
# Запуск: ./play.sh [UUID]

BASE_URL="http://127.0.0.1:5000"
GAME_ID="${1:-$(python3 -c 'import uuid; print(uuid.uuid4())')}"

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

echo -e "${YELLOW}🎮 Крестики-Нолики${NC}"
echo "Игра ID: $GAME_ID"
echo "Управление: координаты 'строка столбец' (0-2)"
echo "Пример: '0 0' — верхний левый, '1 1' — центр"
echo "Команды: 'quit' — выход, 'board' — показать доску"
echo -e "Легенда: ${GREEN}1 = ты (X)${NC}, ${RED}2 = комп (O)${NC}, 0 = пусто"
echo "=================================================="

CURRENT_MATRIX='[[0,0,0],[0,0,0],[0,0,0]]'
GAME_OVER=false

# Функция печати доски
print_board() {
    local matrix="$1"
    echo ""
    echo "    0   1   2"
    echo "  -----------"
    
    python3 -c "
import json
m = json.loads('$matrix')
for i, row in enumerate(m):
    line = []
    for cell in row:
        if cell == 0: line.append(' . ')
        elif cell == 1: line.append('$GREEN X $NC')
        else: line.append('$RED O $NC')
    print(f'{i} |' + '|'.join(line) + '|')
    print('  -----------')
"
    echo ""
}

# 🔥 НАДЁЖНАЯ проверка победителя
check_winner() {
    python3 -c "
import json, sys
m = json.loads('$1')
lines = [
    [(0,0),(0,1),(0,2)], [(1,0),(1,1),(1,2)], [(2,0),(2,1),(2,2)],
    [(0,0),(1,0),(2,0)], [(0,1),(1,1),(2,1)], [(0,2),(1,2),(2,2)],
    [(0,0),(1,1),(2,2)], [(0,2),(1,1),(2,0)]
]
for l in lines:
    c = [m[r][c] for r, c in l]
    if c[0] != 0 and c[0] == c[1] == c[2]:
        print(c[0]); sys.exit()
print(0)
" 2>/dev/null
}

# 🔥 НАДЁЖНАЯ проверка ничьей
check_draw() {
    python3 -c "
import json
m = json.loads('$1')
print('YES' if all(cell != 0 for row in m for cell in row) else 'NO')
" 2>/dev/null
}

# Функция отправки хода
make_move() {
    local row=$1
    local col=$2
    
    # Обновляем матрицу: ставим 1 (игрок)
    NEW_MATRIX=$(python3 -c "
import json, sys
try:
    m = json.loads('$CURRENT_MATRIX')
    if m[$row][$col] != 0:
        print('OCCUPIED')
        sys.exit(0)
    m[$row][$col] = 1
    print(json.dumps(m))
except Exception as e:
    print('ERROR:' + str(e))
" 2>/dev/null)
    
    if [[ "$NEW_MATRIX" == "OCCUPIED" ]]; then
        echo -e "${RED}❌ Клетка занята!${NC} Попробуй другую."
        return 1
    elif [[ "$NEW_MATRIX" == ERROR* ]]; then
        echo -e "${RED}❌ Ошибка парсинга: ${NEW_MATRIX#ERROR:}${NC}"
        return 1
    fi
    
    # Отправляем запрос на сервер
    RESPONSE=$(curl -s -X POST "$BASE_URL/game/$GAME_ID" \
      -H "Content-Type: application/json" \
      -d "{
        \"id\": \"$GAME_ID\",
        \"board\": {
          \"matrix\": $NEW_MATRIX,
          \"size\": 3
        }
      }" 2>/dev/null)
    
    # Проверяем ответ
    if ! echo "$RESPONSE" | python3 -c "import sys,json; json.load(sys.stdin)" >/dev/null 2>&1; then
        echo -e "${RED}❌ Ошибка: сервер вернул некорректный ответ${NC}"
        echo "   Ответ: ${RESPONSE:0:100}..."
        return 1
    fi
    
    # Обновляем матрицу из ответа сервера
    CURRENT_MATRIX=$(echo "$RESPONSE" | python3 -c "import sys,json; print(json.dumps(json.load(sys.stdin)['board']['matrix']))" 2>/dev/null)
    
    if [[ -z "$CURRENT_MATRIX" ]]; then
        echo -e "${RED}❌ Не удалось обновить доску${NC}"
        return 1
    fi
    
    # Показываем обновлённую доску
    print_board "$CURRENT_MATRIX"
    
    # 🔥 ПРОВЕРКА РЕЗУЛЬТАТА (с очисткой от переносов строк)
    WINNER=$(check_winner "$CURRENT_MATRIX" | tr -d '[:space:]')
    
    if [[ "$WINNER" == "1" ]]; then
        echo -e "${GREEN}🎉 ПОБЕДА! Ты выиграл!${NC}"
        echo "Поздравляем! 🏆"
        GAME_OVER=true
        return 2
    elif [[ "$WINNER" == "2" ]]; then
        echo -e "${RED}😔 Компьютер выиграл.${NC}"
        echo "Попробуй ещё раз! 🎮"
        GAME_OVER=true
        return 2
    fi
    
    # Если нет победителя — проверяем ничью
    DRAW=$(check_draw "$CURRENT_MATRIX" | tr -d '[:space:]')
    if [[ "$DRAW" == "YES" ]]; then
        echo -e "${YELLOW}🤝 Ничья! Поле заполнено.${NC}"
        GAME_OVER=true
        return 2
    fi
    
    return 0
}

# Показываем начальную доску
print_board "$CURRENT_MATRIX"

# Главный цикл
while true; do
    if [[ "$GAME_OVER" == "true" ]]; then
        echo ""
        echo "Начни новую: ./play.sh"
        break
    fi

        echo -e "${RED}❌ Вводи числа 0-2. Пример: '0 1'${NC}"
        continue
    fi
    
    make_move "$row" "$col"
    result=$?
    
    if [[ $result -eq 2 ]]; then
        break
    fi
done

TIMEOUT = 10

# === ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ ===
server_process = None


def cleanup_server():
    """Корректно останавливает тестовый сервер."""
    global server_process
    if server_process and server_process.poll() is None:
        print("\n🛑 Остановка тестового сервера...")
        server_process.terminate()
        try:
            server_process.wait(timeout=5)
        except subprocess.TimeoutExpired:
            server_process.kill()


atexit.register(cleanup_server)


def is_port_in_use(host: str, port: int) -> bool:
    """Проверяет, занят ли порт."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        return s.connect_ex((host, port)) == 0


def kill_process_on_port(port: int):
    """Пытается убить процесс, занимающий порт (Linux)."""
    try:
        result = subprocess.run(
            f"lsof -ti :{port}",
            shell=True, capture_output=True, text=True
        )
        if result.stdout.strip():
            pids = result.stdout.strip().split('\n')
            for pid in pids:
                print(f"⚠️  Порт {port} занят PID {pid}. Завершаем...")
                os.kill(int(pid), signal.SIGTERM)
            time.sleep(1)
    except Exception:
        pass


def wait_for_server(max_attempts: int = 30, delay: float = 0.5) -> bool:
    """Ждёт, пока сервер начнёт принимать соединения на порт."""
    for i in range(max_attempts):
        if not is_port_in_use(HOST, PORT):
            time.sleep(delay)
            continue
        return True
    return False


def start_server():
    """Запускает Flask-сервер."""
    global server_process
    
    if is_port_in_use(HOST, PORT):
        print(f"⚠️  Порт {PORT} уже занят. Пытаемся освободить...")
        kill_process_on_port(PORT)
        if is_port_in_use(HOST, PORT):
            print(f"❌ Не удалось освободить порт {PORT}.")
            print(f"   1) Останови другие процессы: kill -9 $(lsof -ti :{PORT})")
            print(f"   2) Или запусти сервер вручную в другом терминале:")
            print(f"      cd {src_dir} && python main.py")
            print(f"   3) Затем перезапусти этот скрипт.")
            sys.exit(1)
    
    print("🚀 Запуск тестового сервера...")
    python_exe = sys.executable
    main_path = os.path.join(src_dir, "main.py")
    
    server_process = subprocess.Popen(
        [python_exe, main_path],
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        cwd=src_dir,
        text=True,
        start_new_session=True
    )
    
    if not wait_for_server():
        print("❌ Сервер не ответил за 15 секунд.")
        if server_process.poll() is None:
            server_process.terminate()
            out, _ = server_process.communicate(timeout=2)
            if out:
                print("📋 Лог сервера:")
                print(out)
        raise RuntimeError("Сервер не запустился.")
    
    print(f"✅ Сервер запущен: {BASE_URL}")


def send_request(url: str, method: str = "POST", data: dict = None, headers: dict = None) -> tuple[int, dict | None]:
    """Отправляет HTTP-запрос и возвращает (статус, ответ)."""
    if headers is None:
        headers = {}
    if data is not None and "Content-Type" not in headers:
        headers["Content-Type"] = "application/json"
        body = json.dumps(data).encode("utf-8")
    else:
        body = data.encode("utf-8") if isinstance(data, str) else None
    
    req = urllib.request.Request(url, data=body, headers=headers, method=method)
    
    try:
        with urllib.request.urlopen(req, timeout=TIMEOUT) as resp:
            response_data = json.loads(resp.read().decode("utf-8"))
            return resp.status, response_data
    except urllib.error.HTTPError as e:
        try:
            error_data = json.loads(e.read().decode("utf-8"))
            return e.code, error_data
        except:
            return e.code, {"error": e.reason}
    except Exception as e:
        return -1, {"error": str(e)}


def count_non_zero(matrix: list) -> int:
    """Считает количество непустых клеток в матрице."""
    return sum(1 for row in matrix for cell in row if cell != 0)

import uuid
from domain.model.game import Game
from domain.model.game_board import GameBoard
from web.model.game_web import GameWeb
from web.model.game_board_web import GameBoardWeb

class GameWebMapper:
    @staticmethod
    def domain_to_web(domain_game: Game) -> GameWeb:
        """Переводит domain-модель в web-модель (для отправки клиенту)."""
        # Здесь всё просто: domain-объекты гарантированно имеют нужные атрибуты
        board_web = GameBoardWeb(
            matrix=domain_game.board.matrix.copy(),
            size=domain_game.board.size
        )
        return GameWeb(id=str(domain_game.id), board=board_web)

    @staticmethod
    def web_to_domain(web_game: GameWeb) -> Game:
        """
        Переводит web-модель в domain-модель (для обработки логики).
        УМЕЕТ работать как с объектами, так и со словарями (dict).
        """
        # --- 1. Извлекаем данные доски (матрицу и размер) ---
        # Проверяем тип: если пришёл словарь (из JSON) или объект (из теста)
        if isinstance(web_game.board, dict):
            # Если словарь — достаём значения по ключам
            matrix = web_game.board['matrix']
            size = web_game.board.get('size', 3)  # get с дефолтом, чтобы не упасть
        else:
            # Если уже объект — достаём через точку
            matrix = web_game.board.matrix
            size = web_game.board.size

        # --- 2. Создаём domain-модель доски ---
        # .copy() обязателен, чтобы не мутировать входящие данные
        board_domain = GameBoard(matrix=matrix.copy(), size=size)

        # --- 3. Создаём domain-модель игры ---
        game = Game(board=board_domain)
        
        # --- 4. Восстанавливаем UUID ---
        # uuid.UUID() умеет принимать и строку, и уже готовый UUID
        if isinstance(web_game.id, str):
            game.id = uuid.UUID(web_game.id)
        else:
            game.id = web_game.id
            
        return game

class GameWeb:
    # id — строка (UUID приходит как строка из HTTP)
    id: str
    # board — объект типа GameBoardWeb
    board: GameBoardWeb

    def to_dict(self) -> dict:
        """Превращает объект в словарь для JSON-сериализации."""
        return {
            'id': self.id,
            'board': {
                'matrix': self.board.matrix,
                'size': self.board.size
            }
        }

def create_game_controller(service: GameServiceInterface) -> Flask:
    """
    Создаёт Flask-приложение с маршрутами для игры.
    :param service: объект, реализующий GameServiceInterface
    :return: настроенный объект Flask
    """
    app = Flask(__name__)

    # =========================================================
    # 🎮 НОВЫЙ МАРШРУТ: Главная страница с веб-интерфейсом
    # =========================================================
    @app.route('/')
    def serve_frontend():
        """Отдаёт HTML-страницу для игры в браузере."""
        static_dir = os.path.join(os.path.dirname(__file__), '..', 'static')
        return send_from_directory(static_dir, 'index.html')
    
    # =========================================================
    # 📦 Маршрут для раздачи статики (CSS/JS, если вынесешь)
    # =========================================================
    @app.route('/static/<path:filename>')
    def serve_static(filename):
        static_dir = os.path.join(os.path.dirname(__file__), '..', 'static')
        return send_from_directory(static_dir, filename)

    # =========================================================
    # 🎯 СУЩЕСТВУЮЩИЙ МАРШРУТ: API для ходов (ИСПРАВЛЕННАЯ ЛОГИКА)
    # =========================================================
    @app.route('/game/<string:game_id>', methods=['POST'])
    def handle_game_move(game_id: str):
        """
        Обработчик хода: принимает ход игрока, возвращает ход компьютера.
        🔥 ПРАВИЛЬНЫЙ ПОРЯДОК: проверяем победу ПОСЛЕ КАЖДОГО хода.
        """
        data = request.get_json()
        
        # 1. Валидация структуры запроса
        if not data or not isinstance(data, dict):
            abort(400, description="Request body must be a valid JSON object")
        
        try:
            # 2. Парсинг в web-модель
            web_game_input = GameWeb(**data)
            # 3. Перевод в domain-модель
            domain_game = GameWebMapper.web_to_domain(web_game_input)
        except (KeyError, ValueError, TypeError) as e:
            abort(400, description=f"Invalid game  {str(e)}")
        
        # =========================================================
        # 🔥 Валидация через репозиторий
        # =========================================================
        previous_board = service.get_previous_board(domain_game.id)
        if previous_board is not None:
            if not service.validate_board(domain_game, previous_board):
                abort(400, description="Board validation failed: previous moves were modified")
        
        # =========================================================
        # 🔥 ПРОВЕРКА ПОБЕДЫ ИГРОКА (сразу после его хода!)
        # =========================================================
        winner = check_winner(domain_game.board.matrix)
        
        if winner == 1:
            # 🎉 ИГРОК ВЫИГРАЛ! Не делаем ход компьютера
            service.save_game_state(domain_game)
            web_game_output = GameWebMapper.domain_to_web(domain_game)
            return jsonify(web_game_output.to_dict()), 200
        
        # Проверяем ничью после хода игрока
        if service.is_game_over(domain_game):
            # 🤝 НИЧЬЯ! Игра окончена
            service.save_game_state(domain_game)
            web_game_output = GameWebMapper.domain_to_web(domain_game)
            return jsonify(web_game_output.to_dict()), 200
        
        # =========================================================
        # 🔥 Ход компьютера (только если игра ПРОДОЛЖАЕТСЯ)
        # =========================================================
        next_move: Tuple[int, int] = service.get_next_move(domain_game)
        
        # Проверяем, не заполнена ли доска
        board_full = all(cell != 0 for row in domain_game.board.matrix for cell in row)
        
        if not board_full and next_move != (-1, -1):
            r, c = next_move
            # Делаем ход только если клетка действительно свободна
            if domain_game.board.matrix[r][c] == 0:
                domain_game.board.matrix[r][c] = 2  # 2 = компьютер
                
                # 🔥 Проверяем победу КОМПЬЮТЕРА после его хода
                winner = check_winner(domain_game.board.matrix)
                if winner == 2:
                    # 😔 КОМПЬЮТЕР ВЫИГРАЛ
                    service.save_game_state(domain_game)
                    web_game_output = GameWebMapper.domain_to_web(domain_game)
                    return jsonify(web_game_output.to_dict()), 200
        
        # Проверяем ничью после хода компьютера
        if service.is_game_over(domain_game):
            # 🤝 НИЧЬЯ
            service.save_game_state(domain_game)
            web_game_output = GameWebMapper.domain_to_web(domain_game)
            return jsonify(web_game_output.to_dict()), 200
        
        # Игра продолжается — сохраняем и возвращаем
        service.save_game_state(domain_game)
        web_game_output = GameWebMapper.domain_to_web(domain_game)
        return jsonify(web_game_output.to_dict()), 200

    return app

  <script>
    const API_BASE = '';
    let gameId = null;
    let board = [];
    let gameActive = true;
    let mySymbol = 1; // 1 = X (игрок), 2 = O (компьютер)

    document.addEventListener('DOMContentLoaded', () => {
      document.getElementById('newGame').onclick = startNewGame;
      startNewGame();
    });

    function startNewGame() {
      gameId = generateUUID();
      board = Array(3).fill(null).map(() => Array(3).fill(0));
      gameActive = true;
      mySymbol = 1; // Всегда начинаем за X
      renderBoard();
      updateStatus('Твой ход (❌)', 'success');
      document.getElementById('gameId').textContent = 'ID: ' + gameId.slice(0, 8) + '...';
    }

    function generateUUID() {
      if (crypto?.randomUUID) return crypto.randomUUID();
      return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, c => {
        const r = Math.random() * 16 | 0, v = c === 'x' ? r : (r & 0x3 | 0x8);
        return v.toString(16);
      });
    }

    function renderBoard() {
      const container = document.getElementById('board');
      container.innerHTML = '';
      
      for (let row = 0; row < 3; row++) {
        for (let col = 0; col < 3; col++) {
          const cell = document.createElement('div');
          cell.className = 'cell';
          cell.dataset.row = row;
          cell.dataset.col = col;
          
          const val = board[row][col];
          if (val === 1) {
            cell.textContent = '❌';
            cell.classList.add('x');
          } else if (val === 2) {
            cell.textContent = '⭕';
            cell.classList.add('o');
          }
          
          // Блокируем клетку если игра окончена или клетка занята
          if (val === 0 && gameActive) {
            cell.style.cursor = 'pointer';
            cell.onclick = () => makeMove(row, col);
          } else {
            cell.style.cursor = 'not-allowed';
          }
          
          container.appendChild(cell);
        }
      }
    }

    async function makeMove(row, col) {
      if (!gameActive || board[row][col] !== 0) return;
      
      // Показываем свой ход СРАЗУ (для отзывчивости)
      board[row][col] = mySymbol;
      renderBoard();
      updateStatus('Думаю...', '');
      
      try {
        const response = await fetch(`${API_BASE}/game/${gameId}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ 
            id: gameId, 
            board: { matrix: board, size: 3 } 
          })
        });
        
        if (!response.ok) {
          const err = await response.json().catch(() => ({}));
          throw new Error(err.description || 'Ошибка сервера');
        }
        
        const data = await response.json();
        
        // === ВАЖНО: Обновляем доску ТОЛЬКО из ответа сервера ===
        board = data.board.matrix;
        renderBoard();
        
        // Проверяем результат игры
        checkGameResult();
        
      } catch (e) {
        console.error(e);
        updateStatus('❌ ' + e.message, 'error');
        // Откат хода при ошибке
        board[row][col] = 0;
        renderBoard();
      }
    }

    function checkGameResult() {
      const winner = checkWinner();
      const isDraw = checkDraw();
      
      if (winner === 1) {
        updateStatus('🎉 Ты победил!', 'winner-text success');
        highlightWinningCells(1);
        gameActive = false;
      } else if (winner === 2) {
        updateStatus('😔 Компьютер выиграл', 'winner-text error');
        highlightWinningCells(2);
        gameActive = false;
      } else if (isDraw) {
        updateStatus('🏁 Ничья!', '');
        gameActive = false;
      } else {
        updateStatus('Твой ход (❌)', 'success');
      }
    }

    function checkWinner() {
      // Все выигрышные комбинации
      const lines = [
        // Горизонтали
        [[0,0], [0,1], [0,2]],
        [[1,0], [1,1], [1,2]],
        [[2,0], [2,1], [2,2]],
        // Вертикали
        [[0,0], [1,0], [2,0]],
        [[0,1], [1,1], [2,1]],
        [[0,2], [1,2], [2,2]],
        // Диагонали
        [[0,0], [1,1], [2,2]],
        [[0,2], [1,1], [2,0]]
      ];
      
      for (const line of lines) {
        const [r1, c1] = line[0];
        const [r2, c2] = line[1];
        const [r3, c3] = line[2];
        
        if (board[r1][c1] !== 0 &&
            board[r1][c1] === board[r2][c2] &&
            board[r2][c2] === board[r3][c3]) {
          return board[r1][c1]; // 1 или 2
        }
      }
      
      return 0; // нет победителя
    }

    function checkDraw() {
      return board.flat().every(cell => cell !== 0);
    }

    function highlightWinningCells(winner) {
      const lines = [
        [[0,0], [0,1], [0,2]],
        [[1,0], [1,1], [1,2]],
        [[2,0], [2,1], [2,2]],
        [[0,0], [1,0], [2,0]],
        [[0,1], [1,1], [2,1]],
        [[0,2], [1,2], [2,2]],
        [[0,0], [1,1], [2,2]],
        [[0,2], [1,1], [2,0]]
      ];
      
      for (const line of lines) {
        const [r1, c1] = line[0];
        const [r2, c2] = line[1];
        const [r3, c3] = line[2];
        
        if (board[r1][c1] === winner &&
            board[r1][c1] === board[r2][c2] &&
            board[r2][c2] === board[r3][c3]) {
          // Подсвечиваем выигрышную линию
          const cells = document.querySelectorAll('.cell');
          cells[r1 * 3 + c1].classList.add('winner');
          cells[r2 * 3 + c2].classList.add('winner');
          cells[r3 * 3 + c3].classList.add('winner');
          break;
        }
      }
    }

    function updateStatus(text, className) {
      const el = document.getElementById('status');
      el.textContent = text;
      el.className = className || '';
    }
